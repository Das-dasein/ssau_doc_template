  <rule id="3" name="authors_and_heading">
    <description>Оформление заголовка и списка авторов зависит от количества авторов.</description>
    <patterns>
      <pattern authors="1">
        <format>Фамилия, И. О. Название [Текст] / И. О. Фамилия. – ...</format>
      </pattern>
      <pattern authors="2-3">
        <format>Первый автор в заголовке, все авторы перечисляются после слеша</format>
        <correct_example>Агафонова, Н. Н. ... / Н. Н. Агафонова, Т. В. Богачев, Л. И. Глушкова.</correct_example>
      </pattern>
      <pattern authors="4+" subtype="есть_ведущий_автор">
        <format>Первый автор в заголовке + [и др.] после слеша</format>
        <correct_example>Бахвалов, Н. С. ... / Н. С. Бахвалов [и др.].</correct_example>
      </pattern>
      <pattern authors="4+" subtype="коллективный_труд">
        <format>Заголовок начинается с названия (без автора), после слеша — первый автор + [и др.]</format>
        <correct_example>История России [Текст]: ... / В. Н. Быков [и др.], отв. ред. В. Н. Сухов.</correct_example>
      </pattern>
    </patterns>
    <comma_rule>Запятая между фамилией и инициалами обязательна ТОЛЬКО в заголовке (Иванов, И. И.). После слеша — без запятой (И. И. Иванов).</comma_rule>
    <foreign_books_rule>Для непереводных иностранных книг ФИО автора пишется ТОЛЬКО латиницей.</foreign_books_rule>
    <error_patterns>
      <pattern>4+ авторов перечислены поимённо вместо [и др.]</pattern>
      <pattern>При 2-3 авторах после слеша указан только первый</pattern>
      <pattern>"и др" без точки или без квадратных скобок</pattern>
      <pattern>Запятая после фамилии стоит после слеша (должна быть только в заголовке)</pattern>
      <pattern>Запятая после фамилии отсутствует в заголовке</pattern>
      <incorrect_example reason="иностранная книга с автором на кириллице">Швабер К. Agile Project Management</incorrect_example>
    </error_patterns>
  </rule>

  <rule id="4" name="publication_source">
    <description>Для законов, постановлений, статей в журналах/газетах после основных сведений идёт источник публикации через двойной слеш.</description>
    <format>// Источник. – Год. – № X. – Ст./С. Y</format>
    <error_patterns>
      <pattern>// отсутствует там, где должен быть</pattern>
      <pattern>Источник публикации не указан для закона/постановления/статьи</pattern>
      <pattern>Неверный порядок элементов после //</pattern>
    </error_patterns>
  </rule>

  <rule id="5" name="url_and_access_date">
    <description>Только для электронных ресурсов. URL и дата обращения идут вместе.</description>
    <format>– URL: https://... (дата обращения: ДД.ММ.ГГГГ).</format>
    <url_formatting>URL — обычным текстом. Без подчёркивания, без markdown-обёрток типа ____https://...____, без переноса строк внутри URL.</url_formatting>
    <error_patterns>
      <pattern>URL без префикса "URL:"</pattern>
      <pattern>Дата обращения без URL или URL без даты обращения</pattern>
      <pattern>URL подчёркнут или обёрнут в markdown-символы</pattern>
      <pattern>Формат даты не ДД.ММ.ГГГГ</pattern>
    </error_patterns>
  </rule>

  <rule id="6" name="volume">
    <description>Формат указания объёма зависит от типа: книга vs статья.</description>
    <book_format position="конец_записи">– 344 с.</book_format>
    <article_format position="после_источника_публикации">– С. 32-42.</article_format>
    <volume_issue_format>Т. 10, № 7 или Вып. 5</volume_issue_format>
    <foreign_equivalents>
      <equivalent context="книга">344 p.</equivalent>
      <equivalent context="статья">P. 32-42</equivalent>
      <equivalent context="том/выпуск">Vol. 10, no. 7</equivalent>
    </foreign_equivalents>
    <error_patterns>
      <pattern>Для книги указан диапазон страниц вместо общего объёма</pattern>
      <pattern>Для статьи указан общий объём вместо диапазона</pattern>
      <pattern>Латинское "c." вместо кириллического "с."</pattern>
      <pattern>Пропущена точка после с./С.</pattern>
    </error_patterns>
  </rule>

  <rule id="7" name="punctuation_and_separators">
    <description>Разные виды дефиса/тире имеют чёткое назначение.</description>
    <element_separator>` – ` (пробел, длинное тире, пробел) — между основными элементами записи</element_separator>
    <compound_word_separator>`-` (дефис без пробелов) — в составных словах и грамматических окончаниях</compound_word_separator>
    <compound_word_examples>
      <example>нормативно-правовой</example>
      <example>2-е изд.</example>
      <example>по-европейски</example>
      <example>ин-т</example>
      <example>ун-та</example>
    </compound_word_examples>
    <identifier_separator>`-` (дефис) — в идентификаторах</identifier_separator>
    <identifier_examples>
      <example>ГОСТ 51771-2001</example>
      <example>№ 1-ФКЗ</example>
      <example>№ 419059-7</example>
    </identifier_examples>
    <error_patterns>
      <pattern>Длинное тире внутри слова или идентификатора</pattern>
      <pattern>Дефис между основными элементами вместо тире</pattern>
      <pattern>Отсутствие пробелов вокруг тире (например, [Текст].–Введ. вместо [Текст]. – Введ.)</pattern>
      <pattern>Двойное тире – –</pattern>
    </error_patterns>
  </rule>

  <rule id="8" name="numbering">
    <description>Номер источника без точки.</description>
    <correct_example>1 Иванов...</correct_example>
    <incorrect_example>1. Иванов...</incorrect_example>
  </rule>

    <rule id="9" name="quotation_marks_in_title">
    <description>Названия источников приводятся без кавычек. В образце кавычки вокруг названий не используются.</description>
    <correct_example>Foundations of Statistical Natural Language Processing [Текст]</correct_example>
    <incorrect_example>«Foundations of Statistical Natural Language Processing» [Текст]</incorrect_example>
    <error_patterns>
      <pattern>Название обёрнуто в кавычки « » или " "</pattern>
    </error_patterns>
  </rule>

  <rule id="10" name="publisher">
    <description>Выходные данные книги включают место, издательство и год в формате "Место: Издательство, Год." Указание только города и страны без издательства — ошибка.</description>
    <correct_example>– М.: Физматлит, 2002. – 630 с.</correct_example>
    <incorrect_examples>
      <incorrect_example reason="нет издательства">– Лондон, Великобритания 2000 – 704 с.</incorrect_example>
      <incorrect_example reason="нет двоеточия и издательства">– Стэнфорд, Калифорния 2008 – 764 с.</incorrect_example>
    </incorrect_examples>
    <applies_to>Книги, монографии, учебные пособия, диссертации, сборники — любые печатные издания с [Текст]</applies_to>
    <does_not_apply_to>Электронные ресурсы без физического издания</does_not_apply_to>
    <error_patterns>
      <pattern>Издательство отсутствует — указан только город (и/или страна) перед годом</pattern>
      <pattern>Между местом и годом нет двоеточия с издательством</pattern>
      <pattern>Страна указана вместе с городом через запятую вместо издательства</pattern>
    </error_patterns>
  </rule>
</rules>

<not_errors>
  <item>Латинские обозначения в иностранных источниках: [Text], [Electronic Resource], N p., P. A-B, Vol. N, no. N</item>
  <item>Точки в кодах специальности и датах диссертаций: 08.01.09.:, 15.02.02.,</item>
  <item>Сокращения, точно соответствующие образцу</item>
</not_errors>

<output_format>
  <type>.docx</type>
  <structure>
    Группировка по правилам 1–10. Для каждой группы, в которой есть найденные ошибки:
    - Заголовок одной строкой: «Правило N. Название»
    - Таблица с тремя колонками: № источника | Фрагмент из работы студента | В чём ошибка (кратко)
  </structure>
  <constraints>
    <constraint>Один источник может попасть в несколько групп — это нормально</constraint>
    <constraint>Не переписывай источник целиком в правильном виде</constraint>
    <constraint>Не объясняй общие правила ГОСТ — только конкретный дефект в конкретном фрагменте</constraint>
    <constraint>Если нельзя однозначно определить, ошибка это или особенность типа документа — выноси в отдельный раздел «Спорные случаи» с указанием, какая информация нужна</constraint>
  </constraints>
</output_format>
